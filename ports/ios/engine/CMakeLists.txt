# The real single-player engine (upstream src/) built for Apple arm64.
# Source lists come from the Windows build so the two never drift; only
# translation units whose whole purpose is a Windows or x86 API are left out
# here, and their Apple replacements live in ports/ios/platform.

include(FetchContent)

get_filename_component(KISAK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(SRC_DIR "${KISAK_ROOT}/src")
set(DEPS_DIR "${KISAK_ROOT}/deps")
set(SCRIPTS_DIR "${KISAK_ROOT}/scripts")

include("${SCRIPTS_DIR}/common_files.cmake")
include("${SCRIPTS_DIR}/sp/sp_files.cmake")

set(KISAK_ENGINE_GROUPS
    SRCROOT DYNENTITY EFFECTSCORE AIM_ASSIST BGAME CGAME CLIENT COMMON DATABASE
    DEVGUI GAME GFX_D3D PHYSICS PHYSICS_ODE QCOMMON RAGDOLL SCRIPT SERVER SOUND
    STRINGED UI UNIVERSAL XANIM ODE ZLIB
)
set(KISAK_ENGINE_SOURCES "")
foreach(group IN LISTS KISAK_ENGINE_GROUPS)
    list(APPEND KISAK_ENGINE_SOURCES ${${group}})
endforeach()
list(FILTER KISAK_ENGINE_SOURCES INCLUDE REGEX "\\.(c|cpp)$")

set(KISAK_ENGINE_WINDOWS_ONLY
    # Miles Sound System backend; KISAK_OPENAL selects snd_driver_openal.cpp.
    "${SRC_DIR}/sound/snd_mss.cpp"
    # Hand-written SSE skinning; r_model_skin.cpp keeps the portable path.
    "${SRC_DIR}/gfx_d3d/r_model_skin_sse.cpp"
    # QueryPerformanceCounter/rdtsc calibration; ports/ios/platform/apple_timing.cpp replaces it.
    "${SRC_DIR}/universal/timing.cpp"
    # Bink SDK playback; ports/ios/engine/cinematic_apple.mm plays the converted
    # video/*.mp4 with AVFoundation instead (see ports/ios/scripts/convert_videos.sh).
    "${SRC_DIR}/gfx_d3d/r_cinematic.cpp"
)
list(REMOVE_ITEM KISAK_ENGINE_SOURCES ${KISAK_ENGINE_WINDOWS_ONLY})

# The win32 group is not built on Apple. These of its files are portable and
# are reused unchanged; ports/ios/engine/apple_sys.cpp replaces the rest.
list(APPEND KISAK_ENGINE_SOURCES
    "${SRC_DIR}/win32/win_localize.cpp"
    "${SRC_DIR}/win32/win_storage.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/apple_sys.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/controller_input.cpp"

    # Replaces src/gfx_d3d/r_cinematic.cpp (Bink); plays converted video/*.mp4.
    "${CMAKE_CURRENT_LIST_DIR}/cinematic_apple.cpp"
    # 64-bit fastfile content loading (engine side of ports/ios/zoneload).
    "${CMAKE_CURRENT_LIST_DIR}/db_zoneload_apple.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/../compat/d3dx9shader_apple.cpp"
    # Direct3D 9 on Apple platforms (headless stage; Metal backend to follow).
    "${CMAKE_CURRENT_LIST_DIR}/../d3d9/d3d9_apple.cpp"
    # Apple Sys_* primitives. Compiled here rather than linked from
    # kisakcod_apple_platform, whose core library duplicates engine objects.
    "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_events.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_timing.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_critical_sections.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_virtual_memory.cpp"
)

# buildnumber.h is generated by the Windows pre-build step; mirror its content.
execute_process(
    COMMAND git rev-list --count HEAD
    WORKING_DIRECTORY "${KISAK_ROOT}"
    OUTPUT_VARIABLE KISAK_BUILD_NUMBER
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
)
if(NOT KISAK_BUILD_NUMBER)
    set(KISAK_BUILD_NUMBER 0)
endif()
set(KISAK_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(CONFIGURE OUTPUT "${KISAK_GENERATED_DIR}/buildnumber.h" CONTENT "#pragma once
#define BUILD_NUMBER ${KISAK_BUILD_NUMBER}

char *__cdecl getBuildNumber();
int getBuildNumberAsInt();
")

# OpenAL Soft: same release as scripts/extern/openal.cmake. It has a CoreAudio
# backend for macOS and iOS and the EFX headers snd_driver_openal.cpp uses.
set(LIBTYPE "STATIC")
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_CONFIG OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_HRTF_DATA OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_AMBDEC_PRESETS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_ENABLE_MODULES OFF CACHE BOOL "" FORCE)
# OpenAL Soft probes for libatomic with a compile check; iOS try-compiles build
# static libraries, so the probe passes without linking and adds a library
# Apple platforms do not have. Their atomics need no extra library.
if(APPLE)
    set(HAVE_LIBATOMIC 0 CACHE INTERNAL "No libatomic on Apple platforms" FORCE)
endif()
FetchContent_Declare(
    openal
    GIT_REPOSITORY https://github.com/kcat/openal-soft.git
    GIT_TAG 1.25.2
    GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(openal)
# Xcode 26 clang rejects the CoreAudio backend's noexcept render callbacks
# under OpenAL Soft's -Werror; the conversion is harmless there.
# Apple SDK headers also test optional Swift feature macros with -Wundef.
# Keep those SDK diagnostics from becoming OpenAL build errors.
target_compile_options(OpenAL PRIVATE -Wno-error=function-effects -Wno-error=undef)

add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../zoneload" "${CMAKE_CURRENT_BINARY_DIR}/zoneload")

# Metal backend for the Direct3D 9 layer. Objective-C++ with ARC, built apart
# from the engine: its forced Windows prelude defines BOOL and other names that
# clash with the Objective-C headers.
enable_language(OBJCXX)
add_library(kisakcod_metal STATIC
    "${CMAKE_CURRENT_LIST_DIR}/../d3d9/metal/dx9_msl_translator.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/../d3d9/metal/metal_backend.mm"
)
target_compile_features(kisakcod_metal PRIVATE cxx_std_20)
target_compile_options(kisakcod_metal PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
# The COMPILE_LANGUAGE generator expression above is ignored by the Xcode generator, so the
# device build compiled this target without ARC: every id store silently skipped its retain
# and Metal objects died at the next autorelease pool pop. Set the Xcode attribute too.
set_target_properties(kisakcod_metal PROPERTIES XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC YES)
target_link_libraries(kisakcod_metal PUBLIC "-framework Metal" "-framework QuartzCore" "-framework Foundation")

# Kept out of the engine libraries for the same reason as kisakcod_metal: the forced Windows
# prelude redefines BOOL and friends, which the Objective-C headers will not tolerate.
add_library(kisakcod_mode STATIC "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_engine_mode.mm")
target_compile_features(kisakcod_mode PRIVATE cxx_std_20)
target_compile_options(kisakcod_mode PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
set_target_properties(kisakcod_mode PROPERTIES XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC YES)
target_link_libraries(kisakcod_mode PUBLIC "-framework Foundation")
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_link_libraries(kisakcod_mode PUBLIC "-framework UIKit")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_link_libraries(kisakcod_metal PUBLIC "-framework UIKit")
endif()

# Multiplayer engine (experimental). Off by default: -DKISAK_BUILD_MP=ON builds it so the
# remaining 64-bit work in the MP-only sources can be measured and worked through.
option(KISAK_BUILD_MP "Also build the multiplayer engine library" OFF)
if(KISAK_BUILD_MP)
    # The SP list was already captured above; clear the groups and re-include so the MP target
    # gets the multiplayer file set (sp_files.cmake appends SP-only sources to the same names).
    foreach(group IN LISTS KISAK_ENGINE_GROUPS)
        unset(${group})
    endforeach()
    unset(CGAME_MP)
    unset(CLIENT_MP)
    unset(GAME_MP)
    unset(SERVER_MP)
    unset(UI_MP)
    include("${SCRIPTS_DIR}/common_files.cmake")
    include("${SCRIPTS_DIR}/mp/mp_files.cmake")
    set(KISAK_MP_GROUPS
        SRCROOT DYNENTITY EFFECTSCORE AIM_ASSIST BGAME CGAME CGAME_MP CLIENT CLIENT_MP COMMON
        DATABASE DEVGUI GAME GAME_MP GFX_D3D PHYSICS PHYSICS_ODE QCOMMON RAGDOLL SCRIPT SERVER
        SERVER_MP SOUND STRINGED UI UI_MP UNIVERSAL XANIM ODE ZLIB
    )
    set(KISAK_MP_SOURCES "")
    foreach(group IN LISTS KISAK_MP_GROUPS)
        list(APPEND KISAK_MP_SOURCES ${${group}})
    endforeach()
    list(FILTER KISAK_MP_SOURCES INCLUDE REGEX "\\.(c|cpp)$")
    list(REMOVE_ITEM KISAK_MP_SOURCES ${KISAK_ENGINE_WINDOWS_ONLY})
    list(REMOVE_DUPLICATES KISAK_MP_SOURCES)
    add_library(kisakcod_engine_mp STATIC ${KISAK_MP_SOURCES}
        "${SRC_DIR}/client_mp/cl_cod4x.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../network/cod4x_transport.cpp"
        "${SRC_DIR}/win32/win_localize.cpp"
        "${SRC_DIR}/win32/win_storage.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/apple_sys.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/controller_input.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/controller_icons.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/cinematic_apple.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/db_zoneload_apple.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../compat/d3dx9shader_apple.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../compat/steam_apple.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../d3d9/d3d9_apple.cpp"
        # Same Apple Sys_* primitives the single-player library links.
        "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_events.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_timing.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_critical_sections.cpp"
        "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_virtual_memory.cpp"
        # UDP sockets and voice stubs; only the multiplayer build needs them.
        "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_net.cpp"
    )
    # HTTP downloads (mods and custom maps) via NSURLSession. Separate target for the same reason
    # as kisakcod_metal/kisakcod_video: the engine's Windows prelude clashes with ObjC headers.
    add_library(kisakcod_download STATIC "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_download.mm"
        "${CMAKE_CURRENT_LIST_DIR}/../platform/apple_master.cpp")
    target_compile_features(kisakcod_download PRIVATE cxx_std_20)
    target_compile_options(kisakcod_download PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
set_target_properties(kisakcod_download PROPERTIES XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC YES)
    target_link_libraries(kisakcod_download PUBLIC "-framework Foundation")
    target_link_libraries(kisakcod_engine_mp PUBLIC kisakcod_download)
    set_source_files_properties("${SRC_DIR}/sound/snd_driver_load_obj.cpp" PROPERTIES
        COMPILE_DEFINITIONS "DR_WAV_IMPLEMENTATION;DR_MP3_IMPLEMENTATION")
    target_compile_features(kisakcod_engine_mp PRIVATE cxx_std_20)
    target_compile_definitions(kisakcod_engine_mp PRIVATE
        KISAK_MP CINEMA USE_SEPARATE_BLIT_TEXTURE KISAK_OPENAL RELEASE_ASSERTS)
    target_include_directories(kisakcod_engine_mp PRIVATE
        "${SRC_DIR}"
        "${DEPS_DIR}"
        "${CMAKE_CURRENT_LIST_DIR}/../compat/native/windows"
        "${CMAKE_CURRENT_LIST_DIR}/../compat/native/directx"
        "${KISAK_GENERATED_DIR}"
    )
    target_compile_options(kisakcod_engine_mp PRIVATE
        "SHELL:-include ${CMAKE_CURRENT_LIST_DIR}/../compat/kisak_apple_prelude.h"
        -fms-extensions -fno-strict-aliasing -fwrapv -w)
    target_link_libraries(kisakcod_engine_mp PUBLIC OpenAL::OpenAL kisakcod_zoneload kisakcod_metal kisakcod_video kisakcod_mode)
endif()

add_library(kisakcod_engine_sp STATIC ${KISAK_ENGINE_SOURCES})
# dr_wav/dr_mp3 implementations live in exactly one translation unit, as in
# scripts/sp/CMakeLists.txt.
set_source_files_properties("${SRC_DIR}/sound/snd_driver_load_obj.cpp" PROPERTIES
    COMPILE_DEFINITIONS "DR_WAV_IMPLEMENTATION;DR_MP3_IMPLEMENTATION")
target_compile_features(kisakcod_engine_sp PRIVATE cxx_std_20)
target_compile_definitions(kisakcod_engine_sp PRIVATE
    KISAK_SP CINEMA USE_SEPARATE_BLIT_TEXTURE KISAK_OPENAL
    # Keep the engine's own invariant checks on while 64-bit issues are found.
    RELEASE_ASSERTS
)
target_include_directories(kisakcod_engine_sp PRIVATE
    "${SRC_DIR}"
    "${DEPS_DIR}"
    "${CMAKE_CURRENT_LIST_DIR}/../compat/native/windows"
    "${CMAKE_CURRENT_LIST_DIR}/../compat/native/directx"
    "${KISAK_GENERATED_DIR}"
)
target_compile_options(kisakcod_engine_sp PRIVATE
    "SHELL:-include ${CMAKE_CURRENT_LIST_DIR}/../compat/kisak_apple_prelude.h"
    -fms-extensions
    # Decompiled code reads through type-punned pointers and relies on
    # two's-complement wraparound, as MSVC happens to allow.
    -fno-strict-aliasing
    -fwrapv
    # The decompiled sources produce tens of thousands of warnings; errors
    # are what matter while the port is brought up.
    -w
)
target_link_libraries(kisakcod_engine_sp PUBLIC OpenAL::OpenAL kisakcod_zoneload kisakcod_metal kisakcod_video kisakcod_mode)
